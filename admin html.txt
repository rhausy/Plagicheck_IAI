        const API_URL = "http://127.0.0.1:8000";

        function initChart() {
            const ctx = document.getElementById('analysesChart').getContext('2d');
            new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim'],
                    datasets: [{
                        label: 'Analyses effectuées',
                        data: [12, 19, 8, 25, 22, 30, 45],
                        borderColor: '#2563eb',
                        borderWidth: 3,
                        backgroundColor: 'rgba(37, 99, 235, 0.05)',
                        fill: true,
                        tension: 0.4
                    }]
                },
                options: {
                    responsive: true,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: { grid: { color: '#f1f5f9' }, beginAtZero: true },
                        x: { grid: { display: false } }
                    }
                }
            });
        }

        async function chargerDashboard() {
            try {
                const resStats = await fetch(API_URL + '/admin/statistiques');
                if (resStats.ok) {
                    const stats = await resStats.json();
                    document.getElementById('kpi-users').innerText = stats.total_utilisateurs;
                    document.getElementById('kpi-reports').innerText = stats.total_rapports;
                    document.getElementById('kpi-analyses').innerText = stats.total_analyses;
                }

                const resSeuil = await fetch(API_URL + '/admin/seuil');
                if (resSeuil.ok) {
                    const donnees = await resSeuil.json();
                    document.getElementById('kpi-threshold').innerText = donnees.seuil + '%';
                    document.getElementById('seuil-field').value = donnees.seuil;
                }

                const resUsers = await fetch(API_URL + '/admin/utilisateurs');
                if (resUsers.ok) {
                    utilisateursCache = await resUsers.json();
                    afficherUtilisateurs(utilisateursCache);
                }
            } catch (err) {
                console.error("Erreur de synchronisation backend:", err);
            }
        }

        function afficherUtilisateurs(liste) {
            const tbody = document.getElementById('user-rows');
            tbody.innerHTML = '';

            if (liste.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" style="text-align:center; color:var(--text-muted); padding:30px;">Aucun utilisateur trouvé.</td></tr>';
                document.getElementById('users-count-text').innerText = "0 compte enregistré";
                return;
            }

            document.getElementById('users-count-text').innerText = liste.length + " compte(s) affiché(s)";

            liste.forEach(function(u) {
                const tr = document.createElement('tr');
                const initiales = (u.nom || u.nom_utilisateur || u.email).substring(0, 2).toUpperCase();

                let btnClass = 'action-btn';

                if (item.decision === null) {
                    // L'encadrant N'A PAS encore statué
                    if (item.score > SEUIL_ALERTE) {
                        decisionHtml = '<span class="badge-decision alerte-auto"><i class="fa-solid fa-triangle-exclamation"></i> Alerte auto (&gt;' + SEUIL_ALERTE + '%)</span>';
                    } else {
                        decisionHtml = '<span class="badge-decision non-evalue">—</span>';
                    }
                } else {
                    // L'encadrant A STATUÉ (Valeur en BDD)
                    btnLabel = '<i class="fa-solid fa-pen"></i> Modifier';
                    btnClass = 'action-btn edit-mode';

                    if (item.decision === 'valide') {
                        decisionHtml = '<span class="badge-decision valide"><i class="fa-solid fa-check"></i> Validé</span>';
                    } else if (item.decision === 'a_revoir') {
                        decisionHtml = '<span class="badge-decision a-revoir"><i class="fa-solid fa-rotate-left"></i> À revoir</span>';
                    } else if (item.decision === 'suspect') {
                        decisionHtml = '<span class="badge-decision suspect-confirme"><i class="fa-solid fa-ban"></i> Suspect confirmé</span>';
                    }
                }

                tr.innerHTML = 
                    '<td>' +
                        '<div class="doc-title-text">' + item.titre + '</div>' +
                        '<div class="doc-author-text">' + item.etudiant + '</div>' +
                    '</td>' +
                    '<td style="color:var(--text-gray); font-weight:600;">' + item.matiere + '</td>' +
                    '<td><span class="score-capsule ' + scoreClass + '">' + item.score + '%</span></td>' +
                    '<td><span style="font-size:12px; color:var(--primary); font-weight:700;"><i class="fa-solid fa-circle-check"></i> ' + item.statut_analyse + '</span></td>' +
                    '<td>' + decisionHtml + '</td>' +
                    '<td style="text-align: right;">' +
                        '<button class="' + btnClass + '" onclick="ouvrirModalEvaluation(' + item.id + ')">' +
                            btnLabel +
                        '</button>' +
                    '</td>';

                tbody.appendChild(tr);
            });
        }

        function filtrerTable(filtre, btn) {
            filtreActif = filtre;
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');

            let resultat = soumissions;
            if (filtre === 'a_evaluer') resultat = soumissions.filter(s => s.decision === null);
            if (filtre === 'alertes') resultat = soumissions.filter(s => s.score > SEUIL_ALERTE);
            if (filtre === 'traites') resultat = soumissions.filter(s => s.decision !== null);

            afficherTableau(resultat);
        }

        function ouvrirModalEvaluation(id) {
            itemEnCoursId = id;
            const item = soumissions.find(s => s.id === id);
            
            document.getElementById('modal-doc-title').innerText = item.titre;
            document.getElementById('modal-student-name').innerText = "Auteur : " + item.etudiant + " — " + item.matiere;
            document.getElementById('modal-score-display').innerText = item.score + "%" + (item.score > SEUIL_ALERTE ? " (Seuil d'alerte dépassé)" : " (Seuil respecté)");
            
            document.getElementById('decision-select').value = item.decision ? item.decision : (item.score > SEUIL_ALERTE ? 'suspect' : 'valide');
            document.getElementById('comment-field').value = item.commentaire || '';

            document.getElementById('modalEvaluation').style.display = 'flex';
        }

        function fermerModal() {
            document.getElementById('modalEvaluation').style.display = 'none';
        }

        function enregistrerDecision() {
            const item = soumissions.find(s => s.id === itemEnCoursId);
            if (item) {
                // Simule l'appel POST /commentaires/{analyse_id}
                item.decision = document.getElementById('decision-select').value;
                item.commentaire = document.getElementById('comment-field').value;
                
                mettreAJourKPI();
                filtrerTable(filtreActif, document.querySelector('.tab-btn.active'));
                fermerModal();
            }
        }
