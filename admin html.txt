        const API_URL = "http://127.0.0.1:8000";

        function initChart() {
            const ctx = document.getElementById('analysesChart').getContext('2d');
            new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim'],
                    datasets: [{
                        label: 'Analyses effectuées',
                        data: [12, 19, 8, 25, 22, 30, 45],
                        borderColor: '#2563eb',
                        borderWidth: 3,
                        backgroundColor: 'rgba(37, 99, 235, 0.05)',
                        fill: true,
                        tension: 0.4
                    }]
                },
                options: {
                    responsive: true,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: { grid: { color: '#f1f5f9' }, beginAtZero: true },
                        x: { grid: { display: false } }
                    }
                }
            });
        }

        async function chargerDashboard() {
            try {
                const resStats = await fetch(API_URL + '/admin/statistiques');
                if (resStats.ok) {
                    const stats = await resStats.json();
                    document.getElementById('kpi-users').innerText = stats.total_utilisateurs;
                    document.getElementById('kpi-reports').innerText = stats.total_rapports;
                    document.getElementById('kpi-analyses').innerText = stats.total_analyses;
                }

                const resSeuil = await fetch(API_URL + '/admin/seuil');
                if (resSeuil.ok) {
                    const donnees = await resSeuil.json();
                    document.getElementById('kpi-threshold').innerText = donnees.seuil + '%';
                    document.getElementById('seuil-field').value = donnees.seuil;
                }

                const resUsers = await fetch(API_URL + '/admin/utilisateurs');
                if (resUsers.ok) {
                    utilisateursCache = await resUsers.json();
                    afficherUtilisateurs(utilisateursCache);
                }
            } catch (err) {
                console.error("Erreur de synchronisation backend:", err);
            }
        }

        function afficherUtilisateurs(liste) {
            const tbody = document.getElementById('user-rows');
            tbody.innerHTML = '';

            if (liste.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" style="text-align:center; color:var(--text-muted); padding:30px;">Aucun utilisateur trouvé.</td></tr>';
                document.getElementById('users-count-text').innerText = "0 compte enregistré";
                return;
            }

            document.getElementById('users-count-text').innerText = liste.length + " compte(s) affiché(s)";

            liste.forEach(function(u) {
                const tr = document.createElement('tr');
                const initiales = (u.nom || u.nom_utilisateur || u.email).substring(0, 2).toUpperCase();
                const roleClass = (u.role || 'etudiant').toLowerCase();

                tr.innerHTML = '<td>' +
                        '<div class="user-cell">' +
                            '<div class="user-avatar">' + initiales + '</div>' +
                            '<div class="user-info">' +
                                '<div class="name">' + (u.nom || u.nom_utilisateur || 'Nom non défini') + '</div>' +
                                '<div class="subtext">ID: #USR-' + u.id + '</div>' +
                            '</div>' +
                        '</div>' +
                    '</td>' +
                    '<td style="color: #475569; font-weight: 500;">' + u.email + '</td>' +
                    '<td>' +
                        '<span class="role-badge ' + roleClass + '">' +
                            '<i class="fa-solid fa-shield-halved" style="font-size:10px;"></i> ' + (u.role || 'Étudiant') +
                        '</span>' +
                    '</td>' +
                    '<td>' +
                        '<span class="status-badge ' + (u.est_actif ? 'active' : 'inactive') + '">' +
                            '<span class="status-dot"></span> ' + (u.est_actif ? 'Actif' : 'Inactif') +
                        '</span>' +
                    '</td>' +
                    '<td>' +
                        '<div class="table-actions">' +
                            '<button class="icon-btn" title="Modifier"><i class="fa-solid fa-pen-to-square"></i></button>' +
                            '<button class="icon-btn ' + (u.est_actif ? 'danger' : '') + '" onclick="toggleStatut(' + u.id + ', ' + (!u.est_actif) + ')" title="' + (u.est_actif ? 'Désactiver' : 'Activer') + '">' +
                                '<i class="fa-solid ' + (u.est_actif ? 'fa-user-xmark' : 'fa-user-check') + '"></i>' +
                            '</button>' +
                        '</div>' +
                    '</td>';
                tbody.appendChild(tr);
            });
        }

        async function toggleStatut(id, nouvelEtat) {
            try {
                const res = await fetch(API_URL + '/admin/utilisateurs/' + id + '/statut?est_actif=' + nouvelEtat, { method: 'PUT' });
                if (res.ok) chargerDashboard();
            } catch (err) {
                console.error("Mise à jour échouée:", err);
            }
        }

        function filtrerParRole(role, btn) {
            roleActif = role;
            document.querySelectorAll('.filter-btn').forEach(function(b) { b.classList.remove('active'); });
            btn.classList.add('active');
            appliquerFiltres();
        }

        function filtrerUtilisateurs(terme) {
            appliquerFiltres(terme);
        }

        function appliquerFiltres(terme) {
            if (terme === undefined) terme = '';
            let liste = utilisateursCache;
            if (roleActif !== 'tous') {
                liste = liste.filter(function(u) {
                    return (u.role || 'Etudiant').toLowerCase() === roleActif.toLowerCase();
                });
            }
            if (terme.trim() !== '') {
                liste = liste.filter(function(u) {
                    return u.email.toLowerCase().includes(terme.toLowerCase()) ||
                        (u.nom && u.nom.toLowerCase().includes(terme.toLowerCase()));
                });
            }
            afficherUtilisateurs(liste);
        }

        async function sauvegarderParametres() {
            const val = document.getElementById('seuil-field').value;
            try {
                const res = await fetch(API_URL + '/admin/seuil?seuil=' + val, { method: 'PUT' });
                if (res.ok) {
                    document.getElementById('kpi-threshold').innerText = val + '%';
                    alert('Seuil mis à jour : ' + val + '%');
                }
            } catch (err) {
                alert("Impossible de contacter le serveur.");
            }
        }

        function ouvrirModal() {
            document.getElementById('modalUser').style.display = 'flex';
        }

        function fermerModal() {
            document.getElementById('modalUser').style.display = 'none';
        }

        async function ajouterUtilisateurManuel() {
            const nom = document.getElementById('new-name').value;
            const email = document.getElementById('new-email').value;
            const role = document.getElementById('new-role').value;

            if (!nom || !email) {
                return alert("Le nom et l'email sont obligatoires");
            }

            try {
                const res = await fetch(API_URL + '/admin/utilisateurs', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        nom: nom,
                        email: email,
                        mot_de_passe: "ChangerMoi123",
                        role: role
                    })
                });

                if (!res.ok) {
                    const erreur = await res.json();
                    return alert("Erreur : " + erreur.detail);
                }

                fermerModal();
                await chargerDashboard();
            } catch (err) {
                alert("Erreur réseau lors de la création de l'utilisateur.");
            }
        }
